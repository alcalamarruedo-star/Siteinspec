name: Publish SiteInspect
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: true
jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/configure-pages@v5
      - name: Prepare complete application
        run: |
          mkdir -p dist/vendor
          cp index.html app.js word.js sw.js manifest.json icon-192.png icon-512.png dist/
          base=https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38
          curl --fail --location --retry 3 "$base/build/pdf.min.mjs" -o dist/vendor/pdf.mjs
          curl --fail --location --retry 3 "$base/build/pdf.worker.min.mjs" -o dist/vendor/pdf.worker.mjs
          curl --fail --location --retry 3 "$base/LICENSE" -o dist/vendor/LICENSE
          test -s dist/vendor/pdf.mjs && test -s dist/vendor/pdf.worker.mjs
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist
      - uses: actions/deploy-pages@v4
        id: deployment
